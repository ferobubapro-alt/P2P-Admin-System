-- 3. ZERO-LAG P2P DETERMINISTIC DELTA-STATE SYNCHRONIZATION ENGINE
-- ============================================================================
local P2P = {
    Seq = 0,
    LastHeartbeat = 0,
    HeartbeatInterval = 1 / 12, -- 12Hz lightweight state packets, 60FPS client lerp
    PeerStates = {},
    ActiveEffects = {},
    InterpolationTargets = {}
}

function P2P:EncodePacket(packetType, payload)
    self.Seq = self.Seq + 1
    return {
        v = "BERKS_P2P_V2",
        seq = self.Seq,
        t = workspace:GetServerTimeNow(),
        senderId = LocalPlayer.UserId,
        senderName = LocalPlayer.Name,
        isB1rkt = IsVerifiedB1rkt(LocalPlayer),
        kind = packetType,
        data = payload
    }
end

function P2P:BroadcastState(packetType, payload)
    local pkt = self:EncodePacket(packetType, payload)
    self.ActiveEffects[packetType .. "_" .. tostring(payload.targetId or 0)] = pkt
    -- Replicate via hidden Character Attribute & ReplicatedStorage bridge for 0-lag P2P
    pcall(function()
        if LocalPlayer.Character then
            LocalPlayer.Character:SetAttribute("BerksP2P_LastPacket", HttpService:JSONEncode(pkt))
        end
    end)
    return pkt
end

function P2P:ForceFullStateSnapshot()
    for key, pkt in pairs(self.ActiveEffects) do
        self:BroadcastState(pkt.kind, pkt.data)
    end
end

-- Continuous 12Hz Delta-Sync loop (Never drops FPS, keeps Polaria/Topk3k/Godzilla 1:1)
RunService.Heartbeat:Connect(function(dt)
    local now = os.clock()
    if now - P2P.LastHeartbeat >= P2P.HeartbeatInterval then
        P2P.LastHeartbeat = now
        if LocalPlayer.Character and LocalPlayer.Character.PrimaryPart then
            local rootCF = LocalPlayer.Character.PrimaryPart.CFrame
            LocalPlayer.Character:SetAttribute("BerksP2P_CFrame", rootCF)
        end
    end
    -- 60FPS smooth client CFrame interpolation for Polaria & Topk3k movers
    for part, targetCF in pairs(P2P.InterpolationTargets) do
        if part and part.Parent then
            part.CFrame = part.CFrame:Lerp(targetCF, math.clamp(dt * 18, 0, 1))
        else
            P2P.InterpolationTargets[part] = nil
        end
    end
end)

local ActiveGodzillaRigs = {}

local function CleanupGodzillaRig(targetPlayer)
    local state = ActiveGodzillaRigs[targetPlayer.UserId]
    if not state then return end
    if state.HeartbeatConn then state.HeartbeatConn:Disconnect() end
    if state.InputConn then state.InputConn:Disconnect() end
    if state.RigModel and state.RigModel.Parent then
        state.RigModel:Destroy()
    end
    -- Restore original character visibility and walkspeed
    if targetPlayer.Character then
        for _, desc in ipairs(targetPlayer.Character:GetDescendants()) do
            if desc:IsA("BasePart") and desc.Name ~= "HumanoidRootPart" then
                desc.Transparency = 0
            end
        end
        local hum = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.WalkSpeed = 16
            hum.JumpPower = 50
            hum.HipHeight = 0
        end
    end
    ActiveGodzillaRigs[targetPlayer.UserId] = nil
end

local function BuildPrimadonStyleGodzillaRig(targetPlayer)
    local char = targetPlayer.Character
    if not char or not char.PrimaryPart then return end
    local root = char.PrimaryPart
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end

    local rig = Instance.new("Model")
    rig.Name = "BerksGodzillaTitan_" .. targetPlayer.Name
    rig.Parent = char

    local function CreateKaijuPart(name, size, color, material)
        local p = Instance.new("Part")
        p.Name = name
        p.Size = size
        p.Color = color or Color3.fromRGB(28, 32, 38)
        p.Material = material or Enum.Material.Slate
        p.CanCollide = false
        p.Massless = true
        p.Parent = rig
        return p
    end

    -- Primadon-style articulated core parts
    local pelvis = CreateKaijuPart("KaijuPelvis", Vector3.new(12, 8, 10))
    local lowerTorso = CreateKaijuPart("KaijuLowerTorso", Vector3.new(15, 14, 12))
    local upperTorso = CreateKaijuPart("KaijuUpperTorso", Vector3.new(17, 16, 13))
    local neck = CreateKaijuPart("KaijuNeck", Vector3.new(8, 9, 8))
    local head = CreateKaijuPart("KaijuHead", Vector3.new(9, 7, 12))
    local jaw = CreateKaijuPart("KaijuJaw", Vector3.new(8, 3.5, 11))

    -- Dorsal Plates (14 Iconic Jagged Godzilla Fins from Tail to Neck that glow Neon Cyan)
    local dorsalFins = {}
    for i = 1, 14 do
        local finScale = 1 + math.sin((i / 14) * math.pi) * 1.4
        local fin = CreateKaijuPart(
            "DorsalPlate_" .. i,
            Vector3.new(1.6, 5.5 * finScale, 4.2 * finScale),
            Color3.fromRGB(38, 48, 60),
            Enum.Material.Cobblestone
        )
        table.insert(dorsalFins, fin)
    end

    -- Multi-segment articulated Tail (8 segments for smooth sine-wave sway & Tail Whip)
    local tailSegments = {}
    for i = 1, 8 do
        local segScale = 1 - (i * 0.09)
        local seg = CreateKaijuPart(
            "TailSeg_" .. i,
            Vector3.new(9 * segScale, 8 * segScale, 9),
            Color3.fromRGB(26, 30, 36),
            Enum.Material.Slate
        )
        table.insert(tailSegments, seg)
    end

    -- Hide default limbs and boost Titan stats
    for _, desc in ipairs(char:GetDescendants()) do
        if desc:IsA("BasePart") and desc.Parent == char and desc.Name ~= "HumanoidRootPart" then
            desc.Transparency = 1
        end
    end
    hum.WalkSpeed = 26
    hum.JumpPower = 85

    local animClock = 0
    local currentAction = "IDLE"
    local dorsalCharge = 0

    -- Procedural 60FPS Primadon/Godzilla Inverse Kinematics & Animation Loop
    local hbConn = RunService.Heartbeat:Connect(function(dt)
        if not char.Parent or not root.Parent then
            CleanupGodzillaRig(targetPlayer)
            return
        end
        animClock = animClock + dt
        local vel = root.AssemblyLinearVelocity
        local horizSpeed = Vector3.new(vel.X, 0, vel.Z).Magnitude
        local baseCF = root.CFrame * CFrame.new(0, 14, 0)

        -- Heavy bipedal stomp bounce when walking
        local walkBob = horizSpeed > 1 and math.sin(animClock * 7) * 1.8 or math.sin(animClock * 2) * 0.4
        local torsoTilt = currentAction == "ATOMIC_BREATH" and math.rad(-18) or math.rad(12)

        pelvis.CFrame = baseCF * CFrame.new(0, walkBob, 0)
        lowerTorso.CFrame = pelvis.CFrame * CFrame.new(0, 10, -1) * CFrame.Angles(torsoTilt, 0, 0)
        upperTorso.CFrame = lowerTorso.CFrame * CFrame.new(0, 13, -1.5) * CFrame.Angles(torsoTilt * 0.6, 0, 0)
        neck.CFrame = upperTorso.CFrame * CFrame.new(0, 10, -2)
        head.CFrame = neck.CFrame * CFrame.new(0, 5, -4)

        local jawOpenAngle = (currentAction == "ATOMIC_BREATH" or currentAction == "TITAN_ROAR") and math.rad(42) or math.rad(4)
        jaw.CFrame = head.CFrame * CFrame.new(0, -3, -1) * CFrame.Angles(jawOpenAngle, 0, 0)

        -- Procedural Sine-Wave Tail Kinematics
        local prevCF = pelvis.CFrame * CFrame.new(0, -1, 5)
        for i, seg in ipairs(tailSegments) do
            local whipYaw = currentAction == "TAIL_WHIP" and (animClock * 16) or (math.sin(animClock * 3 - i * 0.55) * 0.22)
            local segCF = prevCF * CFrame.Angles(math.rad(-4), whipYaw, 0) * CFrame.new(0, 0, 7.5)
            seg.CFrame = segCF
            prevCF = segCF
        end

        -- Update Dorsal Plates & Atomic Charge Neon Color
        for i, fin in ipairs(dorsalFins) do
            fin.CFrame = upperTorso.CFrame * CFrame.new(0, 10 - i * 2.1, 6.5) * CFrame.Angles(math.rad(-20), 0, 0)
            if dorsalCharge >= (i / #dorsalFins) * 100 then
                fin.Material = Enum.Material.Neon
                fin.Color = Color3.fromRGB(0, 210, 255)
            else
                fin.Material = Enum.Material.Cobblestone
                fin.Color = Color3.fromRGB(38, 48, 60)
            end
        end
    end)

    -- Ability Keybinds for the Godzilla Player (Z: Atomic Breath, X: Nuclear Pulse, C: Tail Whip, V: Roar)
    local inputConn
    if targetPlayer == LocalPlayer then
        inputConn = UserInputService.InputBegan:Connect(function(inp, gp)
            if gp then return end
            if inp.KeyCode == Enum.KeyCode.Z then
                currentAction = "ATOMIC_BREATH"
                dorsalCharge = 100
                P2P:BroadcastState("GODZILLA_ABILITY", { targetId = targetPlayer.UserId, ability = "ATOMIC_BREATH" })
                task.delay(3.5, function()
                    currentAction = "IDLE"
                    dorsalCharge = 0
                end)
            elseif inp.KeyCode == Enum.KeyCode.X then
                currentAction = "NUCLEAR_PULSE"
                dorsalCharge = 100
                P2P:BroadcastState("GODZILLA_ABILITY", { targetId = targetPlayer.UserId, ability = "NUCLEAR_PULSE" })
                task.delay(2, function()
                    currentAction = "IDLE"
                    dorsalCharge = 0
                end)
            elseif inp.KeyCode == Enum.KeyCode.C then
                currentAction = "TAIL_WHIP"
                P2P:BroadcastState("GODZILLA_ABILITY", { targetId = targetPlayer.UserId, ability = "TAIL_WHIP" })
                task.delay(1.5, function() currentAction = "IDLE" end)
            elseif inp.KeyCode == Enum.KeyCode.V then
                currentAction = "TITAN_ROAR"
                P2P:BroadcastState("GODZILLA_ABILITY", { targetId = targetPlayer.UserId, ability = "TITAN_ROAR" })
                task.delay(2.2, function() currentAction = "IDLE" end)
            end
        end)
    end

    ActiveGodzillaRigs[targetPlayer.UserId] = {
        RigModel = rig,
        HeartbeatConn = hbConn,
        InputConn = inputConn
    }
end

local function BuildOrGetPsychiatricHospitalMap()
    if HospitalMapInstance and HospitalMapInstance.Parent then
        return HospitalMapInstance:GetPivot().Position

    local origin = Vector3.new(0, 2400, 0)
    local model = Instance.new("Model")

    floor.Position = origin
    floor.Anchored = true
    floor.Material = Enum.Material.Concrete
    floor.Color = Color3.fromRGB(32, 35, 38)
    floor.Parent = model

    local ceiling = floor:Clone()
    ceiling.Name = "CorridorCeiling"
    ceiling.Position = origin + Vector3.new(0, 18, 0)
    ceiling.Parent = model

    -- Interactive Poltergeist Props (Wheelchairs, Chairs, Doors, Medical Trays)
    local propsFolder = Instance.new("Folder")
    propsFolder.Name = "PoltergeistProps"
    propsFolder.Parent = model

    for z = -100, 100, 25 do
        local chair = Instance.new("Part")
        chair.Name = "HospitalChair"
        chair.Size = Vector3.new(4, 4.5, 4)
        chair.Position = origin + Vector3.new(14, 3.2, z)
        chair.Anchored = true
        chair.Material = Enum.Material.Metal

        chair.Parent = propsFolder

        local door = Instance.new("Part")
        door.Name = "WardDoor"
        door.Size = Vector3.new(1, 12, 6)
        door.Position = origin + Vector3.new(-18, 7, z)
        door.Anchored = true
        door.Material = Enum.Material.WoodPlanks
        door.Color = Color3.fromRGB(58, 42, 34)
        door.Parent = propsFolder
    end

    model.Parent = Workspace
    HospitalMapInstance = model
    return origin + Vector3.new(0, 5, 0)
end

local function StopHallucinations(targetPlayer)
    local state = ActiveHallucinations[targetPlayer.UserId]
    if not state then return end
    state.Running = false
    if state.GazeConn then state.GazeConn:Disconnect() end
    if state.ColorCorrection then state.ColorCorrection:Destroy() end
    if state.Blur then state.Blur:Destroy() end
    if targetPlayer == LocalPlayer then
        targetPlayer.CameraMode = Enum.CameraMode.Classic
    end
    ActiveHallucinations[targetPlayer.UserId] = nil
end

local function ComputeSmartHallucinationCFrame(rootPart, spawnMode)
    local dist = math.random(12, 95) -- Smart 1-100 stud range
    local rootCF = rootPart.CFrame
    if spawnMode == 1 then
        -- FRONT FLASH: Directly ahead, vanishes the instant looked at
        return rootCF * CFrame.new(0, 0, -dist) * CFrame.Angles(0, math.pi, 0), "FRONT_FLASH"
    elseif spawnMode == 2 then
        -- BEHIND STALKER: Spawns behind player, vanishes as soon as player turns around
        return rootCF * CFrame.new(0, 0, math.clamp(dist * 0.45, 6, 35)), "BEHIND_STALKER"
    elseif spawnMode == 3 then
        -- CORNER PEEKER: Uses Raycast to find walls/obstacles within 1-100 studs
        return rootCF * CFrame.new(math.random(-28, 28), 0, -dist), "CORNER_PEEKER"
    elseif spawnMode == 4 then
        -- CEILING CRAWLER: Above player looking down
        return rootCF * CFrame.new(0, 14, -math.clamp(dist * 0.5, 8, 40)) * CFrame.Angles(math.pi, 0, 0), "ABOVE_CEILING"
    elseif spawnMode == 5 then
        -- RANDOM SERVER PLAYER SPRINT CLONE
        return rootCF * CFrame.new(-22, 0, -math.clamp(dist * 0.6, 10, 50)), "SERVER_CLONE_RUSH"
    else
        -- SHADOW FIGURE WITH GLOWING EYES
        return rootCF * CFrame.new(math.random(-35, 35), 0, -dist), "SHADOW_EYES"
    end
end

local function StartHallucinationsEngine(casterPlayer, targetPlayer, config)
    StopHallucinations(targetPlayer)
    local state = {
        Running = true,
        HospitalMode = config.hospitalMode or false,
        Multiplier = config.multiplier or 1.0,
        ControlsInverted = false
    }
    ActiveHallucinations[targetPlayer.UserId] = state

    -- Force First-Person until command is removed
    if targetPlayer == LocalPlayer then
        targetPlayer.CameraMode = Enum.CameraMode.LockFirstPerson
        local cc = Instance.new("ColorCorrectionEffect")
        cc.Name = "BerksHallucinationCC"
        cc.Contrast = 0.35 * state.Multiplier
        cc.Saturation = -0.55
        cc.Parent = Lighting
        state.ColorCorrection = cc
    end

    task.spawn(function()
        -- If in Hospital Mode, wait exactly 30 seconds after teleport before hallucinations start!
        if config.initialDelay and config.initialDelay > 0 then
            task.wait(config.initialDelay)
        end

        while state.Running and targetPlayer.Parent do
            local char = targetPlayer.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")
            if root then
                local modeIdx = math.random(1, 6)
                local spawnCF, modeTag = ComputeSmartHallucinationCFrame(root, modeIdx)

                -- Pick either the caster or a random player currently in the server
                local allPlayers = Players:GetPlayers()
                local chosenActor = (modeTag == "SERVER_CLONE_RUSH" and #allPlayers > 0)
                    and allPlayers[math.random(1, #allPlayers)]
                    or casterPlayer

                -- Periodically invert player movement controls for 1 second
                if math.random() < (0.45 * state.Multiplier) then
                    state.ControlsInverted = true
                    task.delay(1.0, function()
                        state.ControlsInverted = false
                    end)
                end

                -- If in Hospital Mode, move hospital furniture (chairs block path, props fall, doors slam)
                if state.HospitalMode and HospitalMapInstance then
                    local props = HospitalMapInstance:FindFirstChild("PoltergeistProps")
                    if props then
                        for _, prop in ipairs(props:GetChildren()) do
                            if (prop.Position - root.Position).Magnitude < 45 then
                                TweenService:Create(prop, TweenInfo.new(0.45, Enum.EasingStyle.Back), {
                                    CFrame = CFrame.new(root.Position + root.CFrame.LookVector * 8) * CFrame.Angles(0, math.rad(math.random(-90, 90)), 0)
                                }):Play()
                            end
                        end
                    end
                end

                P2P:BroadcastState("HALLUCINATION_WAVE", {
                    targetId = targetPlayer.UserId,
                    actorName = chosenActor and chosenActor.Name or "ShadowEntity",
                    mode = modeTag,
                    hospital = state.HospitalMode
                })
            end

            -- Repeat every 10-40 seconds (or 5-20 seconds in 2x Hospital Mode)
            local minWait = math.floor(10 / state.Multiplier)
            local maxWait = math.floor(40 / state.Multiplier)
            task.wait(math.random(minWait, maxWait))
        end
    end)
end

Commands["godzilla"] = function(sender, args)

        CleanupGodzillaRig(target)

        P2P:BroadcastState("HALLUCINATIONS_STATE", { targetId = target.UserId, active = false })
        return
    end
    StartHallucinationsEngine(sender, target, {
        hospitalMode = false,
        multiplier = 1.0,
        initialDelay = 0
    })
end

Commands["hallucinationshospital"] = function(sender, args)
    local target = Players:FindFirstChild(args[1] or sender.Name) or sender
    local hospitalPos = BuildOrGetPsychiatricHospitalMap()
    if target.Character and target.Character.PrimaryPart then
        target.Character:PivotTo(CFrame.new(hospitalPos))
    end
    if sender.Character and sender.Character.PrimaryPart and sender ~= target then
        sender.Character:PivotTo(CFrame.new(hospitalPos + Vector3.new(4, 0, 4)))
    end
    -- Starts after 30 seconds at 2x intensity with moving hospital props!
    StartHallucinationsEngine(sender, target, {
        hospitalMode = true,
        multiplier = 2.0,
        initialDelay = 30
    })
end
